cmake_minimum_required(VERSION 3.15)
project(graphene-md-gpu CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(OpenMP)
add_library(graphene-md-gpu SHARED src/plugin.cpp)
target_include_directories(graphene-md-gpu PRIVATE src)
set_target_properties(graphene-md-gpu PROPERTIES OUTPUT_NAME graphene-md-gpu PREFIX ""
  LIBRARY_OUTPUT_DIRECTORY $<1:${CMAKE_CURRENT_SOURCE_DIR}>
  RUNTIME_OUTPUT_DIRECTORY $<1:${CMAKE_CURRENT_SOURCE_DIR}>)
if(OpenMP_CXX_FOUND)
  target_link_libraries(graphene-md-gpu PRIVATE OpenMP::OpenMP_CXX)
endif()
if(NOT MSVC)
  target_compile_options(graphene-md-gpu PRIVATE -O3)
endif()
